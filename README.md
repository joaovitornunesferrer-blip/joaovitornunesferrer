<script>
  // Animação Digitação com texto "Olá, eu sou João Vítor!"
  const text = "Olá, eu sou João Vítor!";
  let i = 0;
  const speed = 130;
  function typeWriter() {
    if (i < text.length) {
      document.getElementById("typing-animation").innerHTML += text.charAt(i);
      i++;
      setTimeout(typeWriter, speed);
    }
  }
  if (document.getElementById("typing-animation")) {
    typeWriter();
  }
</script>
